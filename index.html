import logging import random import sqlite3 from telegram import Update from telegram.ext import ApplicationBuilder, CommandHandler, ContextTypes # Enable logging logging.basicConfig(format="%(asctime)s - %(name)s - %(levelname)s - %(message)s", level=logging.INFO) # --- DATABASE SETUP --- def init_db(): conn = sqlite3.connect("reaper_scores.db") cursor = conn.cursor() cursor.execute(""" CREATE TABLE IF NOT EXISTS scores ( user_id INTEGER PRIMARY KEY, username TEXT, points INTEGER DEFAULT 0 ) """) conn.commit() conn.close() init_db() # --- REAPER ECOSYSTEM TRIVIA BANK --- TRIVIA_QUESTIONS = [ { "question": "What structural element in the ecosystem represents tiered growth and support layers for projects?", "options": ["Ascension Pillars", "Red Market Candles", "Slippage Walls", "Burn Zones"], "answer": 0 # Index of the correct option }, { "question": "When processing rewards or unlocking ecosystem assets, what fund backs value and secures project floors?", "options": ["The Slush Fund", "The Permanent DRIP Fund", "The Gas Reservoir", "The Validator Pool"], "answer": 1 }, { "question": "How does the Reaper ecosystem typically handle unvalued, lagging, or dead tokens?", "options": ["By minting more supply", "By ignoring them completely", "By reaping, buying out, and utilizing floor mechanisms", "By locking them in liquidity pools forever"], "answer": 2 }, { "question": "Which blockchain is the native home of the Reaper (RPR) token and its market-regulation framework?", "options": ["Ethereum", "Solana", "XRP Ledger (XRPL)", "Bitcoin"], "answer": 2 } ] # Store active trivia state in memory temporarily active_quiz = {} # --- COMMANDS --- async def start(update: Update, context: ContextTypes.DEFAULT_TYPE): await update.message.reply_text( "💀 Welcome to the **Capital Revival Trivia Bot**!\n\n" "Test your knowledge on Ascension Pillars, Boxes, and XRPL mechanics.\n" "Commands:\n" "/trivia - Drop a new ecosystem question\n" "️/leaderboard - View top community reapers" ) async def send_trivia(update: Update, context: ContextTypes.DEFAULT_TYPE): chat_id = update.effective_chat.id q_data = random.choice(TRIVIA_QUESTIONS) # Send native Telegram Quiz poll message = await context.bot.send_poll( chat_id=chat_id, question=q_data["question"], options=q_data["options"], type="quiz", correct_option_id=q_data["answer"], is_anonymous=False, explanation="Keep studying the ecosystem lore and whitepaper!" ) # Save poll ID to track results if needed active_quiz[message.poll.id] = q_data async def leaderboard(update: Update, context: ContextTypes.DEFAULT_TYPE): conn = sqlite3.connect("reaper_scores.db") cursor = conn.cursor() cursor.execute("SELECT username, points FROM scores ORDER BY points DESC LIMIT 5") top_users = cursor.fetchall() conn.close() if not top_users: await update.message.reply_text("🏆 No scores recorded yet! Run /trivia to start earning points.") return text = "🏆 **Top Ascension Reapers (Leaderboard)** 🏆\n\n" for i, (uname, pts) in enumerate(top_users, 1): name = uname if uname else "Anonymous Reaper" text += f"{i}. {name} — {pts} pts\n" await update.message.reply_text(text) # --- MAIN APP --- def main(): # Replace 'YOUR_BOT_TOKEN' with the token you get from @BotFather app = ApplicationBuilder().token("YOUR_BOT_TOKEN").build() app.add_handler(CommandHandler("start", start)) app.add_handler(CommandHandler("trivia", send_trivia)) app.add_handler(CommandHandler("leaderboard", leaderboard)) print("Reaper Trivia Bot is active...") app.run_polling() if __name__ == "__main__": main()
