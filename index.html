// script.js - Updated with Hold and Nudge Mechanics for Sonic Slots const symbols = ['🍒', '🍋', '🔔', '⭐', '💎', '7️⃣']; const symbolValues = { '🍒': 10, '🍋': 20, '🔔': 30, '⭐': 40, '💎': 50, '7️⃣': 100 }; let reels = [0, 0, 0]; let held = [false, false, false]; let nudgesRemaining = 0; let credits = 100; let isSpinning = false; // DOM Elements const reelElements = [document.getElementById('reel1'), document.getElementById('reel2'), document.getElementById('reel3')]; const holdElements = [document.getElementById('hold1'), document.getElementById('hold2'), document.getElementById('hold3')]; const nudgeElements = [document.getElementById('nudge1'), document.getElementById('nudge2'), document.getElementById('nudge3')]; const creditDisplay = document.getElementById('credits'); const messageDisplay = document.getElementById('message'); const spinButton = document.getElementById('spinBtn'); // Initialize Hold Button Listeners holdElements.forEach((btn, index) => { btn.addEventListener('click', () => { if (isSpinning) return; held[index] = !held[index]; btn.classList.toggle('held', held[index]); btn.textContent = held[index] ? 'HELD' : `Hold ${index + 1}`; }); }); // Initialize Nudge Button Listeners (Bumps a specific reel down one symbol) nudgeElements.forEach((btn, index) => { btn.addEventListener('click', () => { if (isSpinning || nudgesRemaining <= 0) return; reels[index] = (reels[index] + 1) % symbols.length; reelElements[index].textContent = symbols[reels[index]]; nudgesRemaining--; messageDisplay.textContent = `Nudge used! Remaining: ${nudgesRemaining}`; if (nudgesRemaining === 0) { disableNudges(); checkWin(); } }); }); spinButton.addEventListener('click', spinReels); function spinReels() { if (isSpinning) return; if (credits < 10) { messageDisplay.textContent = "Not enough credits!"; return; } credits -= 10; creditDisplay.textContent = credits; isSpinning = true; messageDisplay.textContent = "Spinning..."; disableNudges(); // Visual spin animation loop let spinInterval = setInterval(() => { reels.forEach((val, i) => { if (!held[i]) { reels[i] = Math.floor(Math.random() * symbols.length); reelElements[i].textContent = symbols[reels[i]]; } }); }, 100); // Stop spin after 1.5 seconds setTimeout(() => { clearInterval(spinInterval); reels.forEach((val, i) => { if (!held[i]) { reels[i] = Math.floor(Math.random() * symbols.length); reelElements[i].textContent = symbols[reels[i]]; } }); isSpinning = false; // Reset holds after spin concludes held = [false, false, false]; holdElements.forEach((btn, i) => { btn.classList.remove('held'); btn.textContent = `Hold ${i + 1}`; }); // 10% chance to trigger a random Nudge reward feature if (Math.random() < 0.10 && nudgesRemaining === 0) { nudgesRemaining = Math.floor(Math.random() * 2) + 1; // 1 to 2 nudges messageDisplay.textContent = `Feature Triggered! ${nudgesRemaining} Nudge(s) Awarded!`; enableNudges(); } else { checkWin(); } }, 1500); } function checkWin() { if (reels[0] === reels[1] && reels[1] === reels[2]) { let winAmount = symbolValues[symbols[reels[0]]] * 5; credits += winAmount; messageDisplay.textContent = `JACKPOT! Won ${winAmount} credits!`; } else if (reels[0] === reels[1] || reels[1] === reels[2] || reels[0] === reels[2]) { let winAmount = 15; credits += winAmount; messageDisplay.textContent = `Small Win! Won ${winAmount} credits!`; } else { messageDisplay.textContent = "Try again!"; } creditDisplay.textContent = credits; } function enableNudges() { nudgeElements.forEach(btn => btn.style.display = 'inline-block'); } function disableNudges() { nudgeElements.forEach(btn => btn.style.display = 'none'); }
